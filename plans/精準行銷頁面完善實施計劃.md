<script>
// 搜尋功能
function openSearch() {
    // 創建搜尋模態框
    const searchModal = document.createElement('div');
    searchModal.className = 'fixed inset-0 bg-black/50 z-50 flex items-start justify-center pt-20';
    searchModal.innerHTML = `
        <div class="bg-white dark:bg-slate-800 rounded-xl w-full max-w-md mx-4 p-4 shadow-2xl">
            <div class="flex items-center gap-3 mb-4">
                <span class="material-symbols-outlined text-slate-400">search</span>
                <input type="text" placeholder="搜尋行銷活動或客群..." 
                       class="flex-1 bg-transparent border-none outline-none text-slate-900 dark:text-white"
                       autofocus>
                <button onclick="this.closest('.fixed').remove()" class="text-slate-400">
                    <span class="material-symbols-outlined">close</span>
                </button>
            </div>
            <div class="text-sm text-slate-500 dark:text-slate-400">
                <p>試著搜尋：高價值客戶、流失預警、新會員</p>
            </div>
        </div>
    `;
    document.body.appendChild(searchModal);
    
    // 點擊背景關閉
    searchModal.addEventListener('click', function(e) {
        if (e.target === this) {
            this.remove();
        }
    });
}
